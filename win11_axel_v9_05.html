function Nota($nombre) {
    if ($notas.ContainsKey($nombre)) { return $notas[$nombre] }
    foreach ($k in $notas.Keys) {
        if ($k.EndsWith("_") -and $nombre.StartsWith($k)) { return $notas[$k] }
    }
    return "(sin nota — revisar/agregar al diccionario)"
}

function Chk-Reg($path, $name, $expected, $label, $cat) {
    $prop = Get-ItemProperty $path -EA SilentlyContinue
    $val  = if ($prop) {$prop.$name} else {$null}
    $pass = ($val -eq $expected)
    $disp = if ($null -eq $val) {"NO CONFIGURADO"} else {$val}
